```ts
import { State, Event, defineMachine } from "@handfish/effstate-v4";
import { Schema } from "effect";

const Idle = State("Idle", {});
const Running = State("Running", { startedAt: Schema.DateFromSelf });

const Start = Event("Start", {});
const Stop = Event("Stop", {});

const stopwatch = defineMachine({
  initialState: Idle.make(),
  initialContext: { elapsed: 0 },
  states: {
    Idle: { on: { Start: () => ({ goto: Running.make({ startedAt: new Date() }) }) } },
    Running: { on: { Stop: () => ({ goto: Idle.make() }) } },
  },
});
```

## Features

<CardGrid stagger>
  <Card title="Schema-First" icon="document">
    Define states and events once with Effect Schema. Constructors, type guards, and
    serialization come for free — no separate context schema to keep in sync.
  </Card>
  <Card title="Effect-Native" icon="rocket">
    Entry/exit effects and `run` streams have full access to Effect's requirements (`R`)
    and error channels. Streams auto-cancel when you leave a state.
  </Card>
  <Card title="Type Safe" icon="approve-check">
    States and events are discriminated unions. Handlers are narrowed by event tag;
    transitions are checked against your state union at compile time.
  </Card>
  <Card title="Lean" icon="puzzle">
    ~500 lines, zero dependencies beyond Effect. Imperative actor API that drops
    straight into React via `useSyncExternalStore`.
  </Card>
</CardGrid>
